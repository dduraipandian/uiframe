Notification

Notification + error modal notifications (Bootstrap based)
Install & Import

Use as an ES module. The library exports a default `notification` instance. 

<!-- Load core and notification packages -->
<script src="../packages/core/dist/core.min.js"></script>
<script src="../packages/notification/dist/notification.min.js"></script>
Quick Start
// Initialize notification (returns a singleton or per-app instance)
const notification = uiframe.notification({ name: "demo" });

// Basic types
notification.info('Informational message');
notification.success('Saved successfully');
notification.warning('Caution: check this');

// Error with modal details
notification.error('An error occurred', 'Error Title', { 
  status: 500, 
  message: 'Server error' 
});
